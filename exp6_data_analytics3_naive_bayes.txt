# cell 2 – importing required libraries

# importing required python libraries

print("shape of dataset:", df.shape)

# cell 6 – dataset information


# displaying dataset information

print(df.info())


---

# cell 7 – checking missing values


# checking missing values in dataset

print(df.isnull().sum())


---

# cell 8 – statistical summary


# generating statistical summary

print(df.describe())


---

# cell 9 – removing unnecessary id column

# removing id column because it is not useful for prediction

df.drop(
    'Id',
    axis=1,
    inplace=True
)

# displaying updated dataset

print(df.head())


---

# cell 10 – checking unique species values

# displaying unique species values

print(df['Species'].unique())


---

# cell 11 – encoding species column

# converting species names into numeric values

# iris-setosa -> 0
# iris-versicolor -> 1
# iris-virginica -> 2

df['Species'] = df['Species'].map({
    'Iris-setosa': 0,
    'Iris-versicolor': 1,
    'Iris-virginica': 2
})

# displaying updated dataset

print(df.head())


---

# cell 12 – selecting input and output variables


# selecting independent variables

x = df.drop('Species', axis=1)

# selecting dependent variable

y = df['Species']

# displaying variables

print(x.head())

print(y.head())


---

# cell 13 – splitting dataset into training and testing data

# splitting dataset into training and testing sets

x_train, x_test, y_train, y_test = train_test_split(
    x,
    y,
    test_size=0.25,
    random_state=42
)

# displaying shapes

print(x_train.shape)
print(x_test.shape)
print(y_train.shape)
print(y_test.shape)


---

# cell 14 – creating naïve bayes model

# creating gaussian naïve bayes model

model = GaussianNB()


---

# cell 15 – training naïve bayes model


# training naïve bayes model

model.fit(x_train, y_train)

# model learns patterns from training data


---

# cell 16 – predicting test results

# predicting species values

y_pred = model.predict(x_test)

# displaying predicted values

print(y_pred)


---

# cell 17 – comparing actual and predicted values

# comparing actual and predicted values

comparison = pd.DataFrame({
    'actual_species': y_test,
    'predicted_species': y_pred
})

print(comparison.head(10))


---

# cell 18 – creating confusion matrix

# creating confusion matrix

cm = confusion_matrix(y_test, y_pred)

# displaying confusion matrix

print(cm)


---

# cell 19 – plotting confusion matrix heatmap

# plotting confusion matrix heatmap using seaborn

plt.figure(figsize=(6,4))

sns.heatmap(
    cm,
    annot=True,
    fmt='d'
)

plt.title("confusion matrix")
plt.xlabel("predicted values")
plt.ylabel("actual values")

plt.show()


---

# cell 20 – extracting tp, fp, tn, fn values

# extracting values from confusion matrix

# for binary metrics demonstration,
# considering iris-setosa as positive class

tp = cm[0][0]

fp = cm[1][0] + cm[2][0]

fn = cm[0][1] + cm[0][2]

tn = cm[1][1] + cm[1][2] + cm[2][1] + cm[2][2]

# displaying values

print("true positive:", tp)

print("false positive:", fp)

print("false negative:", fn)

print("true negative:", tn)


---

# cell 21 – calculating accuracy

# calculating accuracy

accuracy = accuracy_score(y_test, y_pred)

print("accuracy:", accuracy)


---

# cell 22 – calculating error rate

# calculating error rate

error_rate = 1 - accuracy

print("error rate:", error_rate)


---

# cell 23 – calculating precision

# calculating precision

precision = tp / (tp + fp)

print("precision:", precision)

---

# cell 24 – calculating recall

# calculating recall

recall = tp / (tp + fn)

print("recall:", recall)


---

# cell 25 – classification report


# displaying classification report

report = classification_report(
    y_test,
    y_pred
)

print(report)

# naïve bayes algorithm

naïve bayes is a supervised machine learning classification algorithm based on probability and bayes theorem.

it assumes:

* all features are independent from each other.

---

# bayes theorem

formula:

P(A|B)= {P(B|A)*P(A)}/{P(B)}

where:

* P(A|B) = probability of event a given b
* P(B|A) = probability of event b given a
* P(A) = probability of event a
* P(B) = probability of event b

---

# gaussian naïve bayes

gaussian naïve bayes is used when:

* input features are continuous numeric values.

example:

* sepal length
* petal width

# confusion matrix

confusion matrix is used to evaluate classification performance.

---

# true positive

model correctly predicts positive class.

---

# true negative

model correctly predicts negative class.

---

# false positive

model incorrectly predicts positive class.

---

# false negative

model incorrectly predicts negative class.

---

# accuracy

accuracy measures overall correct predictions.

formula:

Accuracy= {TP+TN}/{TP+TN+FP+FN}

---

# error rate

error rate measures incorrect predictions.

formula:

Error Rate= {FP+FN}/{TP+TN+FP+FN}

---

# precision

precision measures correctness of positive predictions.

formula:

Precision={TP}/{TP+FP}

---

# recall

recall measures how many actual positive cases were correctly identified.

formula:

Recall={TP}/{TP+FN}
