# creating sample document

document = """

healthcare and banking industries use artificial intelligence to improve efficiency.

text analytics helps extract useful information from textual data.
preprocessing methods include tokenization, stop word removal,
stemming, lemmatization, and pos tagging.

# displaying document

print(document)


---

# cell 5 – tokenization

# tokenization splits document into words

tokens = word_tokenize(document)

# displaying tokens

print(tokens)


---

# cell 6 – counting total tokens

# counting total number of tokens

print("total number of tokens:", len(tokens))


---

# cell 7 – pos tagging

# performing part of speech tagging

pos_tags = pos_tag(tokens)

# displaying pos tagged words

print(pos_tags)


---

# cell 8 – removing stop words

# loading english stop words

stop_words = set(stopwords.words('english'))

# removing stop words from tokens

filtered_words = []

for word in tokens:
    
    if word.lower() not in stop_words:
        
        filtered_words.append(word)

# displaying filtered words

print(filtered_words)

---

# cell 9 – counting filtered words

# counting words after stop word removal

print("number of filtered words:", len(filtered_words))

---

# cell 10 – stemming

# creating stemming object

stemmer = PorterStemmer()

# applying stemming

stemmed_words = []

for word in filtered_words:
    
    stemmed_words.append(
        stemmer.stem(word)
    )

# displaying stemmed words

print(stemmed_words)

---

# cell 11 – lemmatization

# creating lemmatizer object

lemmatizer = WordNetLemmatizer()

# applying lemmatization

lemmatized_words = []

for word in filtered_words:
    
    lemmatized_words.append(
        lemmatizer.lemmatize(word)
    )

# displaying lemmatized words

print(lemmatized_words)


---

# cell 12 – comparing stemming and lemmatization

# comparing original, stemmed and lemmatized words

comparison = pd.DataFrame({
    'original_word': filtered_words[:20],
    'stemmed_word': stemmed_words[:20],
    'lemmatized_word': lemmatized_words[:20]
})

print(comparison)

---

# cell 13 – calculating term frequency

# calculating term frequency manually

term_frequency = {}

for word in filtered_words:
    
    word = word.lower()
    
    if word in term_frequency:
        
        term_frequency[word] += 1
        
    else:
        
        term_frequency[word] = 1

# displaying term frequencies

print(term_frequency)


---

# cell 14 – converting term frequency into dataframe

# converting term frequency dictionary into dataframe

tf_df = pd.DataFrame(
    term_frequency.items(),
    columns=['word', 'term_frequency']
)

# sorting by highest frequency

tf_df = tf_df.sort_values(
    by='term_frequency',
    ascending=False
)

# displaying dataframe

print(tf_df.head(20))

# cell 16 – calculating tf-idf

# creating tf-idf vectorizer object

vectorizer = TfidfVectorizer()

# fitting and transforming documents

tfidf_matrix = vectorizer.fit_transform(documents)

# converting tf-idf matrix into dataframe

tfidf_df = pd.DataFrame(
    tfidf_matrix.toarray(),
    columns=vectorizer.get_feature_names_out()
)

# displaying tf-idf values

print(tfidf_df)

# theory

# what is text analytics?

text analytics is the process of extracting meaningful information from textual data using natural language processing and machine learning techniques.

text analytics is used in:

* chatbots
* sentiment analysis
* search engines
* recommendation systems
* spam filtering

---

# natural language processing

natural language processing (nlp) is a branch of artificial intelligence that enables computers to understand and process human language.

---

# text preprocessing

text preprocessing cleans and prepares textual data before analysis.

it improves:

* accuracy
* efficiency
* quality of machine learning models

---

# tokenization

tokenization splits text into smaller units called tokens.

example:

sentence:

```text
machine learning is powerful
```

tokens:

```text
['machine', 'learning', 'is', 'powerful']
```

---

# part of speech tagging

pos tagging identifies grammatical role of words.

examples:

* noun
* verb
* adjective

example:

```text
learning -> noun
run -> verb
beautiful -> adjective
```

---

# stop words

stop words are common words that do not add important meaning.

examples:

* is
* the
* and
* of

these words are removed to reduce unnecessary data.

---

# stemming

stemming reduces words to root form.

examples:

| original word | stemmed word |
| ------------- | ------------ |
| playing       | play         |
| studying      | studi        |
| connected     | connect      |

stemming may produce incomplete words.

---

# lemmatization

lemmatization converts words into meaningful base words.

examples:

| original word | lemmatized word |
| ------------- | --------------- |
| running       | run             |
| studies       | study           |
| better        | good            |

lemmatization produces valid dictionary words.

---

# term frequency

term frequency measures how many times a word appears in a document.

formula:

TF(t)= number of occurrences of term / total number of terms

higher tf means word appears frequently.

---

# inverse document frequency

idf measures importance of word across multiple documents.

formula:

IDF(t)= log({N}/{df(t)})

where:

* N = total number of documents
* df(t) = number of documents containing term t

common words receive lower idf values.

---

# tf-idf

tf-idf combines term frequency and inverse document frequency.

formula:

TF-IDF = TF x IDF

tf-idf helps identify important words in documents.
